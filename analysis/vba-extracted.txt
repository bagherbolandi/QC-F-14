Sub Sabt_Form()

    ' Validate required Data Validation cells
    If WorksheetFunction.CountA(Sheets("FORM").Range("C5,C7,C9,C11,E5,E7,E9,E11,G5")) < 9 Then
        MsgBox "Please complete all required fields before submitting.", _
               vbCritical, "Validation Error"
        Exit Sub
    End If

    ' ==============================
    ' NO MESSAGE IF FORM IS COMPLETE

'
' AUTO Macro
'

'
    Range("C15:C23").Select
    Selection.Copy
    Sheets("AUTO").Select
    Range("A1").Select
    Selection.PasteSpecial Paste:=xlPasteValues, Operation:=xlNone, SkipBlanks _
        :=False, Transpose:=True
    Application.CutCopyMode = False
    Selection.Copy
    Sheets("Data").Select
    Range("A2").Select
    Selection.Insert Shift:=xlDown
    Sheets("FORM").Select
    ActiveWindow.SmallScroll Down:=-11
    Application.CutCopyMode = False
    Range("E11").Select
    Selection.ClearContents
    Range("E9").Select
    Selection.ClearContents
   
End Sub
